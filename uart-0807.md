```rust
// UMWAIT：C0.2 深度休眠，有限超时作为安全网
            // 读取当前 TSC，加 ~10ms（假设 TSC ~3GHz → 30,000,000 ticks）
            let deadline = {
                let lo: u32;
                let hi: u32;
                unsafe {
                    core::arch::asm!(
                        "rdtsc",
                        out("eax") lo,
                        out("edx") hi,
                        options(nomem, nostack),
                    );
                }
                ((hi as u64) << 32 | lo as u64) + 30_000_000
            };
            unsafe {
                umwait(0, deadline);
            }
```

```rust
        loop {
            self.drain_queue();
            if self.pending.get() == 0 {
                break;
            }
            if signal::TERM.load(Ordering::Relaxed) {
                break;
            }
           
            unsafe {
                crate::uintr::syscall::clui();
            }
            if !self.queue.is_empty() {
                unsafe {
                    crate::uintr::syscall::stui();
                }
                continue;
            }
            unsafe {
                umonitor(&WAKE_FLAG as *const _ as *const u8);
            }
            if !self.queue.is_empty() {
                unsafe {
                    crate::uintr::syscall::stui();
                }
                continue;
            }
            unsafe {
                crate::uintr::syscall::stui();
            }
            unsafe {
                umwait(0, u64::MAX);
            }
        }
```

```rust
loop {
            self.drain_queue();
            if self.pending.get() == 0 {
                break;
            }
            if signal::TERM.load(Ordering::Relaxed) {
                break;
            }
            // ── UMONITOR/UMWAIT 低功耗休眠 ──────────────────────────
            
            unsafe {
                crate::uintr::syscall::clui();
            }
            if !self.queue.is_empty() {
                unsafe {
                    crate::uintr::syscall::stui();
                }
                continue;
            }


            unsafe {
                umonitor(&WAKE_FLAG as *const _ as *const u8);
                umwait(0, u64::MAX);
                crate::uintr::syscall::stui();  // stui 放在umonitor前面后面，或者umwait后面都可以
            }
```
